6. FAST SQL SEARCH QUERIES
-------------------------------------------------------------------
-- Fast Pet Pickup Lookup Query (Used by Pet Pickup Search Bar):
SELECT 
  p.pet_id, p.pet_name, p.species, p.breed, 
  c.customer_name AS owner_name, c.contact_number AS owner_contact,
  app.authorized_person, app.contact_number AS authorized_contact, app.authorization_status
FROM pet p
JOIN customer c ON p.customer_id = c.customer_id
LEFT JOIN authorized_pet_pickup app ON p.pet_id = app.pet_id
WHERE p.pet_id = ?;
